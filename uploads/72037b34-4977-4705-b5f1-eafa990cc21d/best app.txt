def build_retriever(pdf_path):
    global retriever, docs

    loader = PyPDFLoader(pdf_path)
    documents = loader.load()

    splitter = RecursiveCharacterTextSplitter(
        chunk_size=2000,
        chunk_overlap=300
    )

    docs = splitter.split_documents(documents)

    vectorstore = FAISS.from_documents(docs, embeddings)
    retriever = vectorstore.as_retriever(search_kwargs={"k": 12})

    print("✅ Document indexed")

def rag_answer(question):

    if retriever is None:
        return "Upload PDF first."

    results = retriever.invoke(question)
    context = " ".join(d.page_content for d in results)

    prompt = f"""
must Answer in ONE SHORT LINE only (max 15 words).
No explanation.

Context:
{context}

Question: {question}
Answer:
"""

    return llm.invoke(prompt).strip()

def generate_summary():

    chunk_summaries = []

    print("Summarizing chunks safely...")

    # STEP 1 — summarize each chunk (very small)
    for d in docs[:12]:   # 🔥 LIMIT chunks (important)
        prompt = f"""
Summarize in ONLY 2 short bullets.

Text:
{d.page_content[:1200]}

Bullets:
"""
        out = llm.invoke(prompt)
        chunk_summaries.append(out)


    # STEP 2 — keep combined text small
    combined = "\n".join(chunk_summaries[:20])


    # STEP 3 — final structured output
    final_prompt = f"""
Using ONLY these points, generate:

SUMMARY:
• 8 bullets

OBLIGATIONS:
• 8 bullets

RISKS:
• 5 bullets

Rules:
- simple English
- full sentences
- MUST include all 3 sections
- do NOT skip any section

Points:
{combined}

Output:
"""

    return llm.invoke(final_prompt)

from sklearn.metrics.pairwise import cosine_similarity

def calculate_similarity(text1, text2):
    emb1 = embeddings.embed_query(text1)
    emb2 = embeddings.embed_query(text2)

    score = cosine_similarity([emb1], [emb2])[0][0]
    return score * 100

    # 1. Depth
    depth = (used_chunks / total_chunks) * 100

    # 2. Same text used for summary
    combined_text = "\n".join(d.page_content[:1000] for d in docs[:used_chunks])

    # 3. Similarity
    similarity = calculate_similarity(out, combined_text)

    # 4. Hybrid
    hybrid = (0.6 * depth) + (0.4 * similarity)

    # 5. Map to 70–95
